## 42. MySQL 数据同步到 Elasticsearch

### 方案对比

| 方案 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| Logstash | 配置简单，支持多种插件 | 资源消耗大，延迟较高 | 数据量小，实时性要求不高 |
| Canal | 轻量级，实时性好 | 需要开启 binlog | 实时性要求高 |
| MQ | 解耦好，可靠性高 | 需要修改业务代码 | 新业务或重构项目 |

### 推荐方案：Canal + Kafka + ES

```java
/**
 * Canal 客户端监听 MySQL binlog
 */
@Component
@Slf4j
public class CanalClient {
    
    @Autowired
    private KafkaTemplate<String, String> kafkaTemplate;
    
    private CanalConnector connector;
    
    @PostConstruct
    public void init() {
        // 连接 Canal Server
        connector = CanalConnectors.newSingleConnector(
            new InetSocketAddress("127.0.0.1", 11111),
            "example", "", ""
        );
        
        // 启动监听线程
        new Thread(this::process).start();
    }
    
    private void process() {
        connector.connect();
        connector.subscribe("shop_db.product");
        connector.rollback();
        
        while (true) {
            try {
                Message message = connector.getWithoutAck(100);
                long batchId = message.getId();
                int size = message.getEntries().size();
                
                if (batchId == -1 || size == 0) {
                    Thread.sleep(1000);
                    continue;
                }
                
                processEntries(message.getEntries());
                connector.ack(batchId);
                
            } catch (Exception e) {
                log.error("Canal process error", e);
                connector.rollback();
            }
        }
    }
    
    private void processEntries(List<Entry> entries) {
        for (Entry entry : entries) {
            if (entry.getEntryType() != EntryType.ROWDATA) {
                continue;
            }
            
            RowChange rowChange = RowChange.parseFrom(entry.getStoreValue());
            EventType eventType = rowChange.getEventType();
            
            for (RowData rowData : rowChange.getRowDatasList()) {
                if (eventType == EventType.INSERT || 
                    eventType == EventType.UPDATE) {
                    handleInsertOrUpdate(rowData);
                } else if (eventType == EventType.DELETE) {
                    handleDelete(rowData);
                }
            }
        }
    }
    
    private void handleInsertOrUpdate(RowData rowData) {
        Map<String, String> dataMap = new HashMap<>();
        
        // 获取变更后的数据
        for (Column column : rowData.getAfterColumnsList()) {
            dataMap.put(column.getName(), column.getValue());
        }
        
        // 发送到 Kafka
        String topic = "product-sync";
        String key = dataMap.get("id");
        String value = JSON.toJSONString(dataMap);
        
        kafkaTemplate.send(topic, key, value);
        log.info("Send to Kafka: {}", value);
    }
    
    private void handleDelete(RowData rowData) {
        // 获取删除前的数据
        for (Column column : rowData.getBeforeColumnsList()) {
            if ("id".equals(column.getName())) {
                String deleteMsg = JSON.toJSONString(
                    Map.of("id", column.getValue(), "operation", "delete")
                );
                kafkaTemplate.send("product-sync", column.getValue(), deleteMsg);
                break;
            }
        }
    }
}
```

### Kafka 消费者同步到 ES

```java
/**
 * 消费 Kafka 消息并同步到 ES
 */
@Component
@Slf4j
public class ProductSyncConsumer {
    
    @Autowired
    private RestHighLevelClient esClient;
    
    @Autowired
    private ProductService productService;
    
    private static final String INDEX_NAME = "product";
    
    @KafkaListener(topics = "product-sync", groupId = "es-sync-group")
    public void consume(ConsumerRecord<String, String> record) {
        try {
            String value = record.value();
            JSONObject data = JSON.parseObject(value);
            
            if ("delete".equals(data.getString("operation"))) {
                deleteFromES(data.getString("id"));
            } else {
                syncToES(data);
            }
            
        } catch (Exception e) {
            log.error("Sync to ES error", e);
            // 可以发送到死信队列重试
        }
    }
    
    private void syncToES(JSONObject data) throws IOException {
        // 从数据库查询完整数据（包括关联表）
        Long productId = data.getLong("id");
        ProductESDTO product = productService.getProductForES(productId);
        
        if (product == null) {
            return;
        }
        
        // 构建索引请求
        IndexRequest request = new IndexRequest(INDEX_NAME)
            .id(String.valueOf(productId))
            .source(JSON.toJSONString(product), XContentType.JSON);
        
        // 执行索引
        IndexResponse response = esClient.index(request, RequestOptions.DEFAULT);
        log.info("Sync to ES success: id={}, result={}", 
                 productId, response.getResult());
    }
    
    private void deleteFromES(String id) throws IOException {
        DeleteRequest request = new DeleteRequest(INDEX_NAME, id);
        DeleteResponse response = esClient.delete(request, RequestOptions.DEFAULT);
        log.info("Delete from ES: id={}, result={}", id, response.getResult());
    }
}
```

### Query DSL 编写

```java
/**
 * 商品搜索服务
 */
@Service
@Slf4j
public class ProductSearchService {
    
    @Autowired
    private RestHighLevelClient esClient;
    
    /**
     * 多维度搜索
     */
    public SearchResult search(SearchParam param) throws IOException {
        // 构建搜索请求
        SearchRequest request = new SearchRequest("product");
        SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
        
        // 1. 构建 bool 查询
        BoolQueryBuilder boolQuery = QueryBuilders.boolQuery();
        
        // 关键词搜索（must）
        if (StringUtils.isNotBlank(param.getKeyword())) {
            boolQuery.must(QueryBuilders.multiMatchQuery(param.getKeyword())
                .field("title", 3.0f)      // 标题权重高
                .field("brand_name", 2.0f)  // 品牌次之
                .field("description", 1.0f) // 描述最低
                .type(MultiMatchQueryBuilder.Type.BEST_FIELDS)
                .operator(Operator.AND));
        }
        
        // 类目筛选（filter）
        if (param.getCategoryId() != null) {
            boolQuery.filter(QueryBuilders.termQuery("category_id", param.getCategoryId()));
        }
        
        // 品牌筛选（filter）
        if (CollectionUtils.isNotEmpty(param.getBrandIds())) {
            boolQuery.filter(QueryBuilders.termsQuery("brand_id", param.getBrandIds()));
        }
        
        // 价格区间（filter）
        if (param.getMinPrice() != null || param.getMaxPrice() != null) {
            RangeQueryBuilder rangeQuery = QueryBuilders.rangeQuery("price");
            if (param.getMinPrice() != null) {
                rangeQuery.gte(param.getMinPrice());
            }
            if (param.getMaxPrice() != null) {
                rangeQuery.lte(param.getMaxPrice());
            }
            boolQuery.filter(rangeQuery);
        }
        
        // 商品属性筛选（nested query）
        if (CollectionUtils.isNotEmpty(param.getAttrs())) {
            for (AttrParam attr : param.getAttrs()) {
                BoolQueryBuilder nestedBool = QueryBuilders.boolQuery()
                    .must(QueryBuilders.termQuery("attrs.attr_id", attr.getAttrId()))
                    .must(QueryBuilders.termsQuery("attrs.attr_value", attr.getAttrValues()));
                
                boolQuery.filter(QueryBuilders.nestedQuery("attrs", nestedBool, ScoreMode.None));
            }
        }
        
        // 只查询上架商品
        boolQuery.filter(QueryBuilders.termQuery("status", 1));
        
        sourceBuilder.query(boolQuery);
        
        // 2. 排序
        String sortField = param.getSortField();
        if (StringUtils.isNotBlank(sortField)) {
            SortOrder order = "asc".equalsIgnoreCase(param.getSortOrder()) 
                ? SortOrder.ASC : SortOrder.DESC;
            sourceBuilder.sort(sortField, order);
        } else {
            // 默认按相关性和销量排序
            sourceBuilder.sort("_score", SortOrder.DESC);
            sourceBuilder.sort("sales", SortOrder.DESC);
        }
        
        // 3. 分页
        int from = (param.getPageNum() - 1) * param.getPageSize();
        sourceBuilder.from(from);
        sourceBuilder.size(param.getPageSize());
        
        // 4. 高亮
        if (StringUtils.isNotBlank(param.getKeyword())) {
            HighlightBuilder highlightBuilder = new HighlightBuilder()
                .field("title")
                .preTags("<span style='color:red'>")
                .postTags("</span>");
            sourceBuilder.highlighter(highlightBuilder);
        }
        
        // 5. 聚合（获取筛选项）
        if (param.isNeedAgg()) {
            // 品牌聚合
            sourceBuilder.aggregation(
                AggregationBuilders.terms("brand_agg")
                    .field("brand_id")
                    .size(100)
                    .subAggregation(
                        AggregationBuilders.topHits("brand_top")
                            .size(1)
                            .fetchSource(new String[]{"brand_name"}, null)
                    )
            );
            
            // 属性聚合
            sourceBuilder.aggregation(
                AggregationBuilders.nested("attr_agg", "attrs")
                    .subAggregation(
                        AggregationBuilders.terms("attr_id_agg")
                            .field("attrs.attr_id")
                            .size(100)
                            .subAggregation(
                                AggregationBuilders.terms("attr_value_agg")
                                    .field("attrs.attr_value")
                                    .size(100)
                            )
                    )
            );
        }
        
        request.source(sourceBuilder);
        
        // 执行搜索
        SearchResponse response = esClient.search(request, RequestOptions.DEFAULT);
        
        // 解析结果
        return parseSearchResponse(response, param.getKeyword());
    }
    
    /**
     * 解析搜索响应
     */
    private SearchResult parseSearchResponse(SearchResponse response, String keyword) {
        SearchResult result = new SearchResult();
        
        SearchHits hits = response.getHits();
        result.setTotal(hits.getTotalHits().value);
        
        // 解析商品列表
        List<ProductESDTO> products = new ArrayList<>();
        for (SearchHit hit : hits.getHits()) {
            ProductESDTO product = JSON.parseObject(
                hit.getSourceAsString(), ProductESDTO.class
            );
            
            // 处理高亮
            if (StringUtils.isNotBlank(keyword)) {
                Map<String, HighlightField> highlightFields = hit.getHighlightFields();
                if (highlightFields.containsKey("title")) {
                    String highlightTitle = highlightFields.get("title")
                        .getFragments()[0].string();
                    product.setTitle(highlightTitle);
                }
            }
            
            products.add(product);
        }
        result.setProducts(products);
        
        // 解析聚合结果
        Aggregations aggregations = response.getAggregations();
        if (aggregations != null) {
            // 品牌聚合
            Terms brandAgg = aggregations.get("brand_agg");
            List<BrandVO> brands = new ArrayList<>();
            for (Terms.Bucket bucket : brandAgg.getBuckets()) {
                TopHits topHits = bucket.getAggregations().get("brand_top");
                SearchHit topHit = topHits.getHits().getHits()[0];
                
                BrandVO brand = new BrandVO();
                brand.setBrandId(Long.parseLong(bucket.getKeyAsString()));
                brand.setBrandName(topHit.getSourceAsMap().get("brand_name").toString());
                brand.setCount(bucket.getDocCount());
                brands.add(brand);
            }
            result.setBrands(brands);
            
            // 属性聚合（类似处理）
            // ...
        }
        
        return result;
    }
}
```

```java
@Service
public class QueryOptimization {
    
    /**
     * 1. filter 代替 query（不需要评分时）
     */
    public void useFilter() {

        // 精确匹配用 filter（会被缓存）

        boolQuery.filter(QueryBuilders.rangeQuery("price").gte(100).lte(500));

     * 2. 控制返回字段
     */
    public void limitFields() {
        SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
        
        // 只返回需要的字段
        sourceBuilder.fetchSource(
            new String[]{"id", "title", "price", "image"}, 
            null
        );
    }
    
    /**
     * 3. 使用 scroll 处理大结果集
     */
    public void useScroll() throws IOException {
        RestHighLevelClient client = null;
        
        SearchRequest request = new SearchRequest("product");
        request.scroll(TimeValue.timeValueMinutes(1));
        SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
        sourceBuilder.size(1000);
        request.source(sourceBuilder);
        
        SearchResponse response = client.search(request, RequestOptions.DEFAULT);
        String scrollId = response.getScrollId();
        
        while (true) {
            SearchScrollRequest scrollRequest = new SearchScrollRequest(scrollId);
            scrollRequest.scroll(TimeValue.timeValueMinutes(1));
            
            SearchResponse scrollResponse = client.scroll(scrollRequest, RequestOptions.DEFAULT);

                break;

            // 处理数据

        }
        
        // 清理 scroll
```

```java
/**
 * Function Score 自定义算分
 */
public SearchSourceBuilder buildFunctionScoreQuery(String keyword) {
    SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
    
    // 基础查询
    MultiMatchQueryBuilder baseQuery = QueryBuilders.multiMatchQuery(keyword)
        .field("title", 3.0f)
        .field("brand_name", 2.0f)
        .field("description", 1.0f);
    
    // Function Score 组合多个评分因子
    FunctionScoreQueryBuilder functionScoreQuery = QueryBuilders.functionScoreQuery(
        baseQuery,
        new FunctionScoreQueryBuilder.FilterFunctionBuilder[]{
            // 1. 销量加权（对数函数避免头部效应）
            new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                ScoreFunctionBuilders.fieldValueFactorFunction("sales")
                    .modifier(FieldValueFactorFunction.Modifier.LOG1P)
                    .factor(0.5f)
            ),
            
            // 2. 评分加权
            new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                ScoreFunctionBuilders.fieldValueFactorFunction("rating")
                    .factor(0.3f)
            ),
            
            // 3. 时间衰减（越新越好）
            new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                ScoreFunctionBuilders.exponentialDecayFunction("create_time", 
                    "now", "30d", "7d")
                    .setWeight(0.2f)
            ),
            
            // 4. 价格区间加权
            new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                QueryBuilders.rangeQuery("price").gte(100).lte(500),
                ScoreFunctionBuilders.weightFactorFunction(1.2f)
            )
        }
    )
    .scoreMode(FunctionScoreQuery.ScoreMode.SUM)    // 分数求和
    .boostMode(CombineFunction.MULTIPLY);            // 与原始分数相乘
    
    sourceBuilder.query(functionScoreQuery);
    return sourceBuilder;
}
```

```java
        SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
        
        // 1. 获取用户画像
        UserProfile profile = getUserProfile(userId);
        
        // 2. 构建查询
        BoolQueryBuilder boolQuery = QueryBuilders.boolQuery();
        
        // 基础搜索条件
        if (StringUtils.isNotBlank(param.getKeyword())) {
            boolQuery.must(QueryBuilders.multiMatchQuery(param.getKeyword())
                .field("title", 3.0f)
                .field("description", 1.0f));
        }
        
        // 3. 个性化加权
        FunctionScoreQueryBuilder functionScoreQuery = QueryBuilders.functionScoreQuery(
            boolQuery,
            buildPersonalizedFunctions(profile)
        )
        .scoreMode(FunctionScoreQuery.ScoreMode.SUM)
        .boostMode(CombineFunction.MULTIPLY);
        
        sourceBuilder.query(functionScoreQuery);
        
        // 4. 排序（结合算分和个性化因素）
        sourceBuilder.sort("_score", SortOrder.DESC);
        
        // 分页
        sourceBuilder.from((param.getPageNum() - 1) * param.getPageSize());
        sourceBuilder.size(param.getPageSize());
        
        SearchRequest request = new SearchRequest("product");
        request.source(sourceBuilder);
        
        SearchResponse response = esClient.search(request, RequestOptions.DEFAULT);
        
        // 记录用户搜索行为
        recordSearchBehavior(userId, param.getKeyword());
        
        return parseSearchResponse(response);
    }
    
    /**
     * 构建个性化评分函数
     */
    private FunctionScoreQueryBuilder.FilterFunctionBuilder[] buildPersonalizedFunctions(
            UserProfile profile) {
        
        List<FunctionScoreQueryBuilder.FilterFunctionBuilder> functions = new ArrayList<>();
        
        // 1. 偏好类目加权
        if (CollectionUtils.isNotEmpty(profile.getFavoriteCategoryIds())) {
            for (Long categoryId : profile.getFavoriteCategoryIds()) {
                functions.add(new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                    QueryBuilders.termQuery("category_id", categoryId),
                    ScoreFunctionBuilders.weightFactorFunction(1.5f)
                ));
            }
        }
        
        // 2. 偏好品牌加权
        if (CollectionUtils.isNotEmpty(profile.getFavoriteBrandIds())) {
            functions.add(new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                QueryBuilders.termsQuery("brand_id", profile.getFavoriteBrandIds()),
                ScoreFunctionBuilders.weightFactorFunction(1.3f)
            ));
        }
        
        // 3. 价格偏好
        if (profile.getAvgPrice() != null) {
            double avgPrice = profile.getAvgPrice();
            functions.add(new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                ScoreFunctionBuilders.gaussDecayFunction("price", avgPrice, 
                    avgPrice * 0.5)  // 偏好价格±50%范围内
            ));
        }
        
        // 4. 协同过滤（浏览过相似商品的用户也喜欢）
        if (CollectionUtils.isNotEmpty(profile.getRecommendedProductIds())) {
            functions.add(new FunctionScoreQueryBuilder.FilterFunctionBuilder(
                QueryBuilders.termsQuery("id", profile.getRecommendedProductIds()),
                ScoreFunctionBuilders.weightFactorFunction(1.2f)
            ));
        }
        
        return functions.toArray(new FunctionScoreQueryBuilder.FilterFunctionBuilder[0]);
    }
    
    /**
     * 获取用户画像
     */
    private UserProfile getUserProfile(Long userId) {
        String key = "user:profile:" + userId;
        UserProfile profile = (UserProfile) redisTemplate.opsForValue().get(key);
        
        if (profile == null) {
            profile = buildUserProfile(userId);
            redisTemplate.opsForValue().set(key, profile,Duration.ofHours(24));
        }
        
        return profile;
    }
    
    /**
     * 构建用户画像
     */
    private UserProfile buildUserProfile(Long userId) {
        UserProfile profile = new UserProfile();
        
        // 从用户行为数据中分析
        // 1. 浏览历史 -> 偏好类目、品牌
        // 2. 购买历史 -> 价格区间偏好
        // 3. 搜索历史 -> 兴趣关键词
        // 4. 收藏/加购 -> 高意向商品
        
        // 这里简化处理，实际可以用机器学习模型
        List<Long> favoriteCategoryIds = getUserFavoriteCategories(userId);
        List<Long> favoriteBrandIds = getUserFavoriteBrands(userId);
        Double avgPrice = getUserAvgPrice(userId);
        List<Long> recommendedIds = getCollaborativeRecommendations(userId);
        
        profile.setUserId(userId);
        profile.setFavoriteCategoryIds(favoriteCategoryIds);
        profile.setFavoriteBrandIds(favoriteBrandIds);
        profile.setAvgPrice(avgPrice);
        profile.setRecommendedProductIds(recommendedIds);
        
        return profile;
    }
    
    /**
     * 记录搜索行为
     */
    private void recordSearchBehavior(Long userId, String keyword) {
        // 记录到 Redis 或 MySQL
        String key = "user:search:history:" + userId;
        redisTemplate.opsForZSet().add(key, keyword, System.currentTimeMillis());
        
        // 保留最近 100 条
        redisTemplate.opsForZSet().removeRange(key, 0, -101);
    }
```

```java
/**
 * 搜索建议服务
 */
@Service
public class SearchSuggestionService {
    
    @Autowired
    private RestHighLevelClient esClient;
    
    /**
     * 自动补全建议
     */
    public List<String> suggest(String prefix) throws IOException {
        SearchRequest request = new SearchRequest("product");
        SearchSourceBuilder sourceBuilder = new SearchSourceBuilder();
        
        // 使用 completion suggester
        SuggestBuilder suggestBuilder = new SuggestBuilder();
        suggestBuilder.addSuggestion("title-suggest",
            SuggestBuilders.completionSuggestion("suggest")
                .prefix(prefix)
                .size(10)
                .skipDuplicates(true)
        );
        
        sourceBuilder.suggest(suggestBuilder);
        request.source(sourceBuilder);
        
        SearchResponse response = esClient.search(request, RequestOptions.DEFAULT);
        
        // 解析建议结果
        Suggest suggest = response.getSuggest();
        CompletionSuggestion suggestion = suggest.getSuggestion("title-suggest");
        
        return suggestion.getOptions().stream()
            .map(option -> option.getText().string())
            .collect(Collectors.toList());
    }
}
```
