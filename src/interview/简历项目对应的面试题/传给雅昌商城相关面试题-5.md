```java
/**
 * 基于 Spring Cloud Gateway 的灰度发布实现
 */
@Configuration
public class GrayReleaseConfig {
    
    @Bean
    public GlobalFilter grayReleaseFilter() {
        return new GrayReleaseFilter();
    }
}

public class GrayReleaseFilter implements GlobalFilter, Ordered {
    
    @Autowired
    private GrayReleaseStrategy grayStrategy;
    
    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        ServerHttpRequest request = exchange.getRequest();
        
        // 1. 获取灰度标识
        String grayFlag = request.getHeaders().getFirst("X-Gray-Flag");
        String userId = request.getHeaders().getFirst("X-User-Id");
        
        // 2. 判断是否走灰度版本
        if (grayStrategy.shouldUseGrayVersion(userId, grayFlag)) {
            // 添加灰度标记到请求头
            ServerHttpRequest modifiedRequest = request.mutate()
                .header("X-Gray-Version", "v2")
                .build();
            
            return chain.filter(exchange.mutate()
                .request(modifiedRequest).build());
        }
        
        return chain.filter(exchange);
    }
    
    @Override
    public int getOrder() {
        return -100;
    }
}

/**
 * 灰度发布策略
 */
@Component
public class GrayReleaseStrategy {
    
    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    
    private static final String GRAY_USER_KEY = "gray:users";
    private static final String GRAY_RATIO_KEY = "gray:ratio";
    
    /**
     * 判断是否使用灰度版本
     */
    public boolean shouldUseGrayVersion(String userId, String grayFlag) {
        // 1. 强制灰度标识
        if ("true".equals(grayFlag)) {
            return true;
        }
        
        // 2. 白名单用户
        if (isGrayUser(userId)) {
            return true;
        }
        
        // 3. 按比例灰度
        return shouldGrayByRatio(userId);
    }
    
    /**
     * 检查是否为灰度用户
     */
    private boolean isGrayUser(String userId) {
        if (userId == null) return false;
        return redisTemplate.opsForSet()
            .isMember(GRAY_USER_KEY, userId);
    }
    
    /**
     * 按比例灰度
     */
    private boolean shouldGrayByRatio(String userId) {
        if (userId == null) return false;
        
        String ratioStr = redisTemplate.opsForValue().get(GRAY_RATIO_KEY);
        int ratio = ratioStr != null ? Integer.parseInt(ratioStr) : 0;
        
        // 根据用户ID哈希值决定
        int hash = Math.abs(userId.hashCode() % 100);
        return hash < ratio;
    }
    
    /**
     * 设置灰度比例（0-100）
     */
    public void setGrayRatio(int ratio) {
        redisTemplate.opsForValue()
            .set(GRAY_RATIO_KEY, String.valueOf(ratio));
    }
    
    /**
     * 添加灰度用户
     */
    public void addGrayUser(String userId) {
        redisTemplate.opsForSet().add(GRAY_USER_KEY, userId);
    }
}
```

```java
/**
 * 回滚管理器
 */
@Service
public class RollbackManager {
    
    @Autowired
    private DeploymentService deploymentService;
    
    @Autowired
    private ConfigService configService;
    
    /**
     * 快速回滚
     */
    public void quickRollback(String appName) {
        try {
            // 1. 获取上一个稳定版本
            String previousVersion = deploymentService
                .getPreviousStableVersion(appName);
            
            // 2. 修改流量配置，切回旧版本
            configService.updateTrafficRule(appName, 
                "stable", 100,
                "canary", 0);
            
            // 3. 记录回滚日志
            log.info("Rollback {} to version {}", 
                appName, previousVersion);
            
            // 4. 发送告警通知
            alertService.sendRollbackAlert(appName, previousVersion);
            
        } catch (Exception e) {
            log.error("Rollback failed", e);
            throw new RollbackException("回滚失败", e);
        }
    }
    
    /**
     * 完整回滚（包括数据库）
     */
    @Transactional
    public void fullRollback(String appName, String targetVersion) {
        // 1. 应用回滚
        deploymentService.rollbackToVersion(appName, targetVersion);
        
        // 2. 配置回滚
        configService.rollbackConfig(appName, targetVersion);
        
        // 3. 数据库回滚（如果需要）
        if (needDatabaseRollback(targetVersion)) {
            databaseService.rollbackSchema(targetVersion);
        }
    }
}
```

### 主动健康检查实现

```java
/**
 * 自定义健康检查服务
 */
@Service
public class HealthCheckService {
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Autowired
    private ServerRegistry serverRegistry;
    
    /**
     * 定时健康检查
     */
    @Scheduled(fixedDelay = 10000)  // 每10秒检查一次
    public void performHealthCheck() {
        List<Server> servers = serverRegistry.getAllServers();
        
        servers.parallelStream().forEach(server -> {
            boolean isHealthy = checkServerHealth(server);
            server.setHealthy(isHealthy);
            
            if (!isHealthy) {
                log.warn("Server {} is unhealthy", server.getUrl());
                handleUnhealthyServer(server);
            }
        });
    }
    
    /**
     * 检查单个服务器健康状态
     */
    private boolean checkServerHealth(Server server) {
        try {
            String healthUrl = server.getUrl() + "/actuator/health";
            
            ResponseEntity<HealthResponse> response = restTemplate.exchange(
                healthUrl,
                HttpMethod.GET,
                null,
                HealthResponse.class
            );
            
            // 检查响应状态和健康指标
            return response.getStatusCode().is2xxSuccessful() 
                && "UP".equals(response.getBody().getStatus());
                
        } catch (Exception e) {
            log.error("Health check failed for {}", server.getUrl(), e);
            return false;
        }
    }
    
    /**
     * 处理不健康的服务器
     */
    private void handleUnhealthyServer(Server server) {
        // 1. 增加失败计数
        server.incrementFailureCount();
        
        // 2. 如果连续失败超过阈值，从服务列表移除
        if (server.getFailureCount() >= 3) {
            serverRegistry.removeServer(server);
            alertService.sendAlert("服务器下线: " + server.getUrl());
        }
        
        // 3. 触发故障转移
        triggerFailover(server);
    }
    
    /**
     * 故障转移
     */
    private void triggerFailover(Server failedServer) {
        // 将流量切换到其他健康的服务器
        List<Server> healthyServers = serverRegistry
            .getHealthyServers();
            
        if (healthyServers.isEmpty()) {
            log.error("No healthy servers available!");
            alertService.sendCriticalAlert("所有服务器不可用");
        }
    }
}

/**
 * 服务器注册中心
 */
@Component
public class ServerRegistry {
    
    private final Map<String, Server> servers = new ConcurrentHashMap<>();
    
    public void registerServer(Server server) {
        servers.put(server.getId(), server);
    }
    
    public void removeServer(Server server) {
        servers.remove(server.getId());
    }
    
    public List<Server> getHealthyServers() {
        return servers.values().stream()
            .filter(Server::isHealthy)
            .collect(Collectors.toList());
    }
    
    public List<Server> getAllServers() {
        return new ArrayList<>(servers.values());
    }
}
```

```java
        (client, ELECTION_PATH, serverId);

            @Override
            public void isLeader() {
                log.info("当前节点成为Leader");
                onBecomeLeader();
            }
            
            @Override
            public void notLeader() {
                log.info("当前节点变为Follower");
                onBecomeFollower();
            }
        });
        
        try {
            leaderLatch.start();
        } catch (Exception e) {
            log.error("Leader选举启动失败", e);
        }
    }

     * 成为 Leader 后的处理
     */
    private void onBecomeLeader() {
        // 1. 启动定时任务
        startScheduledTasks();
        
        // 2. 接管集群管理
        takeoverClusterManagement();
        
        // 3. 发送通知
        notifyBecomeLeader();
    }
    
    /**
     * 成为 Follower 后的处理
     */
    private void onBecomeFollower() {
        // 1. 停止定时任务
        stopScheduledTasks();
        
        // 2. 释放资源
        releaseResources();
    }
    
    /**
     * 检查是否为 Leader
     */
    public boolean isLeader() {
        return leaderLatch.hasLeadership();
    }
    
    /**
     * 获取当前 Leader
     */
    public String getCurrentLeader() throws Exception {
        Participant leader = leaderLatch.getLeader();
        return leader != null ? leader.getId() : null;
    }
    
    @PreDestroy
    public void destroy() {
        try {
            if (leaderLatch != null) {
                leaderLatch.close();
            }
            if (client != null) {
                client.close();
            }
        } catch (IOException e) {

            }
```

### 案例2：Redis 缓存雪崩

```java
/**
 * 问题：大量缓存同时失效导致数据库压力剧增
 * 
 * 解决方案：
 * 1. 缓存预热
 * 2. 失效时间随机化
 * 3. 多级缓存
 * 4. 熔断降级
 */
@Service
@Slf4j
public class ImprovedCacheService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Autowired
    private ProductRepository productRepository;
    
    private final LoadingCache<String, Object> localCache = 
        Caffeine.newBuilder()
            .maximumSize(10000)
            .expireAfterWrite(5, TimeUnit.MINUTES)
            .build(key -> null);
    
    /**
     * 改进的缓存获取方法
     */
    public Product getProduct(Long productId) {
        String cacheKey = "product:" + productId;
        
        // 1. 本地缓存
        Product product = (Product) localCache.getIfPresent(cacheKey);
        if (product != null) {
            return product;
        }
        
        // 2. Redis 缓存
        product = (Product) redisTemplate.opsForValue().get(cacheKey);
        if (product != null) {
            localCache.put(cacheKey, product);
            return product;
        }
        
        // 3. 防止缓存击穿：分布式锁
        String lockKey = "lock:" + cacheKey;
        Boolean lockAcquired = redisTemplate.opsForValue()
            .setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);
        
        try {
            if (Boolean.TRUE.equals(lockAcquired)) {
                // 双重检查
                product = (Product) redisTemplate.opsForValue().get(cacheKey);
                if (product != null) {
                    return product;
                }
                
                // 从数据库加载
                product = productRepository.findById(productId).orElse(null);
                
                if (product != null) {
                    // 随机过期时间：防止雪崩
                    long expireTime = 3600 + ThreadLocalRandom.current().nextInt(600);
                    redisTemplate.opsForValue().set(
                        cacheKey, product, expireTime, TimeUnit.SECONDS);
                    
                    localCache.put(cacheKey, product);
                } else {
                    // 缓存空值：防止缓存穿透
                    redisTemplate.opsForValue().set(
                        cacheKey, new NullValue(), 300, TimeUnit.SECONDS);
                }
                
                return product;
            } else {
                // 等待锁释放后重试
                Thread.sleep(50);
                return getProduct(productId);
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new RuntimeException("获取数据失败", e);
        } finally {
            if (Boolean.TRUE.equals(lockAcquired)) {
                redisTemplate.delete(lockKey);
            }
        }
    }
    
    /**
     * 缓存预热
     */
    @PostConstruct
    public void warmUpCache() {
        log.info("开始缓存预热...");
        
        CompletableFuture.runAsync(() -> {
            // 预热热门商品
            List<Product> hotProducts = productRepository.findHotProducts();
            hotProducts.forEach(product -> {
                String cacheKey = "product:" + product.getId();
                long expireTime = 3600 + ThreadLocalRandom.current().nextInt(600);
                redisTemplate.opsForValue().set(
                    cacheKey, product, expireTime, TimeUnit.SECONDS);
            });
            
            log.info("缓存预热完成，预热商品数: {}", hotProducts.size());
        });
    }
}
```

### 案例3：服务雪崩与熔断降级

```java
/**
 * 问题：下游服务超时导致整个调用链阻塞
 * 
 * 解决方案：Resilience4j 熔断器
 */
@Service
@Slf4j
public class OrderService {
    
    @Autowired
    private PaymentServiceClient paymentClient;
    
    @Autowired
    private InventoryServiceClient inventoryClient;
    
    private final CircuitBreaker circuitBreaker;
    private final Retry retry;
    private final Bulkhead bulkhead;
    
    public OrderService() {
        // 熔断器配置
        CircuitBreakerConfig cbConfig = CircuitBreakerConfig.custom()
            .failureRateThreshold(50)                    // 失败率阈值50%
            .slowCallRateThreshold(50)                   // 慢调用阈值50%
            .slowCallDurationThreshold(Duration.ofSeconds(2))
            .waitDurationInOpenState(Duration.ofSeconds(30))
            .permittedNumberOfCallsInHalfOpenState(10)
            .slidingWindowSize(100)
            .minimumNumberOfCalls(10)
            .build();
        
        circuitBreaker = CircuitBreaker.of("paymentService", cbConfig);
        
        // 重试配置
        RetryConfig retryConfig = RetryConfig.custom()
            .maxAttempts(3)
            .waitDuration(Duration.ofMillis(500))
            .retryExceptions(TimeoutException.class, IOException.class)
            .build();
        
        retry = Retry.of("paymentService", retryConfig);
        
        // 舱壁隔离配置
        BulkheadConfig bulkheadConfig = BulkheadConfig.custom()
            .maxConcurrentCalls(25)
            .maxWaitDuration(Duration.ofMillis(500))
            .build();
        
        bulkhead = Bulkhead.of("paymentService", bulkheadConfig);
        
        // 监听熔断器事件
        circuitBreaker.getEventPublisher()
            .onStateTransition(event -> 
                log.warn("熔断器状态变更: {}", event))
            .onError(event ->
                log.error("熔断器捕获错误: {}", event.getThrowable().getMessage()));
    }
    
    /**
     * 创建订单（带熔断保护）
     */
    public OrderResult createOrder(OrderRequest request) {
        // 组合多个保护机制
        Supplier<OrderResult> supplier = () -> doCreateOrder(request);
        
        supplier = Bulkhead.decorateSupplier(bulkhead, supplier);
        supplier = Retry.decorateSupplier(retry, supplier);
        supplier = CircuitBreaker.decorateSupplier(circuitBreaker, supplier);
        
        try {
            return supplier.get();
        } catch (CallNotPermittedException e) {
            log.warn("熔断器开启，执行降级逻辑");
            return fallbackCreateOrder(request);
        } catch (BulkheadFullException e) {
            log.warn("舱壁已满，拒绝请求");
            throw new ServiceUnavailableException("系统繁忙，请稍后重试");
        }
    }
    
    /**
     * 实际业务逻辑
     */
    private OrderResult doCreateOrder(OrderRequest request) {
        // 1. 扣减库存
        inventoryClient.decreaseStock(request.getProductId(), 
            request.getQuantity());
        
        // 2. 创建支付订单
        PaymentResult payment = paymentClient.createPayment(request);
        
        // 3. 创建订单记录
        Order order = new Order();
        order.setPaymentId(payment.getPaymentId());
        orderRepository.save(order);
        
        return OrderResult.success(order);
    }
    
    /**
     * 降级方法
     */
    private OrderResult fallbackCreateOrder(OrderRequest request) {
        // 降级策略：
        // 1. 记录订单到消息队列，异步处理
        messageQueue.send("order.fallback", request);
        
        // 2. 返回友好提示
        return OrderResult.fallback(
            "系统繁忙，订单已提交，请稍后查询");
    }
}
```
